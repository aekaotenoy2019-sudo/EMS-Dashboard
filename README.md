function decisionModel(rows,target=80,closure=10){
 const known=rows.filter(d=>d.n!==null),missing=rows.filter(d=>d.n===null);
 const sum=(arr,f)=>arr.reduce((s,d)=>s+f(d),0);
 const t=sum(rows,d=>d.t),kt=sum(known,d=>d.t),n=sum(known,d=>d.n),gap=kt-n;
 const y68=sum(rows,d=>d.y68),y69=sum(rows,d=>d.y69);
 const need=sum(known,d=>Math.max(0,Math.ceil(d.t*target/100)-d.n));
 const projected=known.map(d=>({...d,add:Math.round((d.t-d.n)*closure/100)}));
 const add=sum(projected,d=>d.add),forecast=sum(rows,d=>d.forecastEMS??Math.max(0,Math.min(d.t,d.y69+d.y69-d.y68)));
 const priorities=[...known].sort((a,b)=>(b.t-b.n)-(a.t-a.n)).slice(0,3);
 return {rows,known,missing,t,kt,n,gap,y68,y69,need,add,forecast,priorities,rate:kt?100*n/kt:null,target,closure};
}
